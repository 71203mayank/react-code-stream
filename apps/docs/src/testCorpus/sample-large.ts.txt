export interface Task0 {
  id: string
  title: string
  priority: number
  tags: readonly string[]
}

/* Module 0: normalize task metadata before scheduling.
   Keep comments and template strings intact across stream boundaries. */
export function scheduleTask0(task: Task0, now: number): string {
  const title = task.title.trim().replace(/\s+/g, ' ')
  const priority = Math.max(0, Math.min(10, task.priority))
  const tags = task.tags.filter((tag) => tag.length > 0).join(', ')
  if (!title) throw new Error('Task requires a title')
  const metadata = { id: task.id, priority, scheduledAt: now }
  return `${metadata.id}: ${title} (${tags}) @ ${metadata.scheduledAt}`
}
