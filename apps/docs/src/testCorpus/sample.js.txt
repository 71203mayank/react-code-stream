const DEFAULT_LIMIT = 8;

/* A tiny seeded generator keeps examples repeatable.
function makeRandom(seed) {
  let value = seed >>> 0;
  return () => {
    value = (value * 1664525 + 1013904223) >>> 0;
    return value / 0x100000000;
  };
}
   Changing the seed produces a different sequence of rows. */

function createRows(count, seed = Date.now()) {
  const random = makeRandom(seed);
  const names = ["cedar", "maple", "birch", "willow", "aspen"];
  return Array.from({ length: count }, (_, index) => ({
    id: index + 1,
    name: names[Math.floor(random() * names.length)],
    score: Math.round(random() * 1000) / 10,
    active: random() > 0.35,
  }));
}

function summarize(rows, limit = DEFAULT_LIMIT) {
  const visible = rows.slice(0, Math.max(0, limit));
  const activeCount = visible.filter((row) => row.active).length;
  const average = visible.length
    ? visible.reduce((total, row) => total + row.score, 0) / visible.length
    : 0;
  return {
    shown: visible.length,
    activeCount,
    average: Number(average.toFixed(2)),
    names: [...new Set(visible.map(({ name }) => name))],
  };
}

const rows = createRows(14, 42);
console.log("Generated sample:", summarize(rows, 10));

module.exports = { createRows, summarize };
